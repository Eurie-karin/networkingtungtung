To update github
1. git pull
	#downloads latest changes
2. git add . 
	#tracks all changes made so far
3. git status
	#ensure everythings okay
4. git commit -m "message"
	#saves snapshot of work, message states the purpose for changing
5. git push
	#uploads to github
